    const aliasKeys = CONFIG_KEY_ALIASES[key] || [];
    for (let i = 0; i < aliasKeys.length; i++) {
      const aliasValue = getStoredValue(aliasKeys[i], CONFIG_MISSING);
      if (aliasValue !== CONFIG_MISSING) return aliasValue;
    }

    return fallback;
  }

  function normalizeConfigBoolean(value) {
    if (value === true || value === 'true' || value === 1 || value === '1') return true;
    if (value === false || value === 'false' || value === 0 || value === '0') return false;
    return null;
  }

  function normalizeTranslatorValue(value) {
    if (!value) return DEFAULT_TRANSLATOR_SETTINGS.engine;
    if (TRANSLATOR_OPTIONS.some(function(option) { return option.value === value; })) return value;
    return TRANSLATOR_LABEL_TO_VALUE[value] || DEFAULT_TRANSLATOR_SETTINGS.engine;
  }

  function normalizeLanguageValue(value, options, fallback) {
    if (!value) return fallback;
    if (options.some(function(option) { return option.value === value; })) return value;
    const mapped = LANGUAGE_LABEL_TO_VALUE[value];
    if (mapped && options.some(function(option) { return option.value === mapped; })) return mapped;
    return fallback;
  }

  function getTranslatorLabel(value) {
    return TRANSLATOR_VALUE_TO_LABEL[value] || value;
  }

  function getLanguageConfigLabel(value) {
    return LANGUAGE_VALUE_TO_LABEL[value] || value;
  }

  function readDefaultTranslate() {
    const legacyRaw = getStoredValue(DEFAULT_SITE_PREF_KEY, undefined);
    const legacy = legacyRaw === 'on' ? true : legacyRaw === 'off' ? false : normalizeConfigBoolean(legacyRaw);
    const configValue = normalizeConfigBoolean(getConfigValue(CONFIG_KEYS.defaultTranslate, undefined));
    const resolved = configValue === null ? legacy : configValue;
    defaultTranslateEnabled = resolved === null ? true : resolved;
    setStoredValue(CONFIG_KEYS.defaultTranslate, defaultTranslateEnabled);
    return defaultTranslateEnabled;
  }

    return floatingTranslateEnabled;

    const configValue = normalizeConfigBoolean(getConfigValue(CONFIG_KEYS.selectionEnabled, undefined));
    const resolved = configValue === null ? legacy : configValue;
    selectionTranslateEnabled = resolved !== false;
    setStoredValue(CONFIG_KEYS.selectionEnabled, selectionTranslateEnabled);
    return selectionTranslateEnabled;

  function normalizeSelectionDirection(value) {
    const settings = Object.assign({}, DEFAULT_SELECTION_DIRECTION, value || {});
    settings.smartDetect = settings.smartDetect === true;
    if (!hasOption(TARGET_LANGUAGE_OPTIONS, settings.langA)) settings.langA = DEFAULT_SELECTION_DIRECTION.langA;
    if (!hasOption(TARGET_LANGUAGE_OPTIONS, settings.langB)) settings.langB = DEFAULT_SELECTION_DIRECTION.langB;
    if (settings.langA === settings.langB) {
      settings.langB = settings.langA === 'zh-Hans' ? 'en' : 'zh-Hans';
    }
    return settings;
  }

  function readSelectionDirection() {
    const legacy = getStoredValue(SELECTION_DIRECTION_KEY, null);
    const legacySettings = legacy && typeof legacy === 'object' && !Array.isArray(legacy) ? legacy : {};
    const defaultA = languageSettings.from === 'auto' ? 'en' : languageSettings.from;
    const defaultB = languageSettings.to || FALLBACK_TARGET_LANG;

    const legacySmartDetect = typeof legacySettings.smartDetect === 'boolean'
      ? legacySettings.smartDetect
      : (typeof legacySettings.bidirectional === 'boolean' ? legacySettings.bidirectional : null);
    const configSmartDetect = normalizeConfigBoolean(getConfigValue(CONFIG_KEYS.selectionSmartDetect, undefined));
    const smartDetect = configSmartDetect === null
      ? (legacySmartDetect === null ? DEFAULT_SELECTION_DIRECTION.smartDetect : legacySmartDetect)
      : configSmartDetect;

    selectionDirection = normalizeSelectionDirection({
      smartDetect: smartDetect,
      langA: normalizeLanguageValue(getConfigValue(CONFIG_KEYS.selectionLangA, legacySettings.langA), TARGET_LANGUAGE_OPTIONS, defaultA),
      langB: normalizeLanguageValue(getConfigValue(CONFIG_KEYS.selectionLangB, legacySettings.langB), TARGET_LANGUAGE_OPTIONS, defaultB)
    });
    persistSelectionDirection();
    return selectionDirection;
  }

  function createUiElement(tag, className, attributes, text, children) {
    const element = document.createElement(tag);
    if (className) element.className = className;
    if (attributes) {
      Object.keys(attributes).forEach(function(name) {
        const value = attributes[name];
        if (name === 'hidden') {
          element.hidden = Boolean(value);
        } else if (value !== null && value !== undefined) {
          element.setAttribute(name, value);
        }
      });
    }
    if (text !== null && text !== undefined) element.textContent = text;
    if (children) children.forEach(function(child) { element.appendChild(child); });
    return element;
  }

  function ensureControlUi() {
    if (controlRoot || !document.body) return;

    injectControlStyle();
    controlRoot = document.createElement('div');
    controlRoot.id = 'auto-translate-control';
    controlRoot.className = 'auto-translate-control no-translate script-element';
    controlRoot.setAttribute('translate', 'no');
    controlRoot.setAttribute('data-no-translate', 'true');
    controlRoot.setAttribute('data-protect-translation', 'true');
    controlRoot.setAttribute('data-auto-translate-ui', 'true');

    const icon = createUiElement('button', 'auto-translate-icon', {
      type: 'button',
      'data-role': 'icon',
      'aria-label': '自动翻译',
      'aria-pressed': 'false',
      title: '点击翻译当前网页，拖动可移动'
    }, '译');
    controlRoot.appendChild(icon);

    document.body.appendChild(controlRoot);
    applyIconOpacity();
    applyStoredControlPosition();

    icon.addEventListener('pointerdown', handleControlPointerDown);
    icon.addEventListener('pointermove', handleControlPointerMove);
    icon.addEventListener('pointerup', handleControlPointerUp);
    icon.addEventListener('pointercancel', handleControlPointerUp);
    icon.addEventListener('click', function(event) {
      if (suppressNextIconClick) {
        event.preventDefault();
        event.stopPropagation();
        suppressNextIconClick = false;
        return;
      }
      toggleTranslation();
    });

    updateControlUi();
    scheduleIconDim();
  }

  function applyStoredControlPosition() {
    const position = readControlPosition();
    if (!controlRoot || !position) return;
    const clamped = clampControlPosition(position.left, position.top);
    controlRoot.style.left = clamped.left + 'px';
    controlRoot.style.top = clamped.top + 'px';
    controlRoot.style.right = 'auto';
  }

  function clampControlPosition(left, top) {
    const size = controlRoot ? Math.max(controlRoot.offsetWidth || 42, 42) : 42;
    const maxLeft = Math.max(8, window.innerWidth - size - 8);
    const maxTop = Math.max(8, window.innerHeight - size - 8);
    return {
      left: Math.min(Math.max(8, left), maxLeft),
      top: Math.min(Math.max(8, top), maxTop)
    };
  }

  function handleControlPointerDown(event) {
    if (event.button !== undefined && event.button !== 0) return;
    if (!controlRoot) return;

    clearIconDim();
    const rect = controlRoot.getBoundingClientRect();
    dragStartX = event.clientX;
    dragStartY = event.clientY;
    dragStartLeft = rect.left;
    dragStartTop = rect.top;
    draggingControl = false;
    suppressNextIconClick = false;

    try { event.currentTarget.setPointerCapture(event.pointerId); } catch (e) {}
  }

  function handleControlPointerMove(event) {
    if (dragStartX === 0 && dragStartY === 0) return;

    const deltaX = event.clientX - dragStartX;
    const deltaY = event.clientY - dragStartY;
    if (!draggingControl && Math.hypot(deltaX, deltaY) < 4) return;

    draggingControl = true;
    suppressNextIconClick = true;

    const position = clampControlPosition(dragStartLeft + deltaX, dragStartTop + deltaY);
    controlRoot.style.left = position.left + 'px';
    controlRoot.style.top = position.top + 'px';
    controlRoot.style.right = 'auto';
    event.preventDefault();
  }

  function handleControlPointerUp(event) {
    if (draggingControl && controlRoot) {
      const rect = controlRoot.getBoundingClientRect();
      const position = clampControlPosition(rect.left, rect.top);
      saveControlPosition(position.left, position.top);
      scheduleIconDim();
    }

    draggingControl = false;
    dragStartX = 0;
    dragStartY = 0;
    try { event.currentTarget.releasePointerCapture(event.pointerId); } catch (e) {}
  }

  function updateControlUi() {
    if (!controlRoot) return;

    const icon = controlRoot.querySelector('[data-role="icon"]');
    controlRoot.dataset.state = translationEnabled ? 'on' : 'off';
    icon.title = translationEnabled
      ? '已翻译，点击还原原文，拖动可移动'
      : '点击翻译当前网页，拖动可移动';
    icon.setAttribute('aria-pressed', translationEnabled ? 'true' : 'false');
  }

  function toggleTranslation() {
    if (translationEnabled) {
      disableTranslation();
    } else {
      enableTranslation();
    }
    scheduleIconDim();
  }

  async function enableTranslation() {
    translationEnabled = true;
    updateControlUi();
    await startTranslationForSite();
  }

  function disableTranslation() {
    translationEnabled = false;
    pageTranslating = false;
    stopDOMObservation();
    stopContinuousTranslation();
    restoreOriginalText();
    updateControlUi();
  }

  function restoreOriginalText() {
    const translatedNodes = document.querySelectorAll('.microsoft-translated[data-original-id]');
    translatedNodes.forEach(function(span) {
      const originalText = originalTextMap.get(span.dataset.originalId);
      if (typeof originalText === 'string' && span.parentNode) {
        span.parentNode.replaceChild(document.createTextNode(originalText), span);
      }
    });
  }

  async function startTranslationForSite() {
    if (translatorStartPromise) return translatorStartPromise;

    translatorStartPromise = (async function() {
      if (!translationEnabled) return;

      await setupActiveTranslator();
      if (!translationEnabled) return;
      if (!translatorReady) {
        translationEnabled = false;
        updateControlUi();
        return;
      }

      scheduleInitialTranslation();
      startDOMObservation();
      startContinuousTranslation();
    })();

    translatorStartPromise.finally(function() {
      translatorStartPromise = null;
    });
    return translatorStartPromise;
  }

  function scheduleInitialTranslation() {
    const run = function() {
      if (translationEnabled) translateWholePage();
    };

    if (document.readyState === 'complete' || document.readyState === 'interactive') {
      setTimeout(run, AUTO_TRANSLATE_DELAY);
    } else {
      window.addEventListener('load', function() {
        setTimeout(run, AUTO_TRANSLATE_DELAY);
      }, { once: true });
    }
  }

  function requestText(options) {
    return new Promise((resolve, reject) => {
      GM_xmlhttpRequest({
        method: options.method || 'GET',
        url: options.url,
        headers: options.headers || {},
        data: options.data,
        onload: function(res) {
          if (res.status >= 200 && res.status < 300) {
            resolve(res.responseText);
          } else {
            reject(new Error('请求失败: ' + res.status));
          }
        },
        onerror: function() {
          reject(new Error('网络请求失败'));
        }
      });
    });
  }

  function normalizeLangForEngine(lang, engine) {
    if (engine === 'microsoft') return lang;
    if (lang === 'zh-Hans') return 'zh-CN';
    if (lang === 'zh-Hant') return 'zh-TW';
    return lang;
  }

  function normalizeSourceLang(fromLang, engine) {
    if (!fromLang || fromLang === 'auto') return engine === 'microsoft' ? null : 'auto';
    return normalizeLangForEngine(fromLang, engine);
  }

  function normalizeTargetLang(toLang, engine) {
    return normalizeLangForEngine(toLang || FALLBACK_TARGET_LANG, engine);
  }

  async function setupActiveTranslator() {
    const engine = translatorSettings.engine;
    translatorReady = false;

    if (engine === 'microsoft') {
      // 新版 Edge 翻译接口免鉴权，不再需要先获取 token。
      translatorReady = true;
      return;
    }

    if (engine === 'custom' && !translatorSettings.customUrl.trim()) {
      console.error('自定义翻译接口为空');
      return;
    }

    translatorReady = true;
  }

  function translateMicrosoftBatch(texts, fromLang, toLang) {
    return new Promise((resolve, reject) => {
      if (!texts || texts.length === 0) {
        return reject(new Error('参数无效'));
      }

      // 微软已下线旧的 edge.microsoft.com/translate/auth 取 token 接口（返回 404）。
      // 新版接口免鉴权，直接提交文本数组；from=auto 会返回 400，所以自动识别时省略 from。
      const params = ['to=' + encodeURIComponent(toLang)];
      if (fromLang) params.push('from=' + encodeURIComponent(fromLang));
      const url = 'https://edge.microsoft.com/translate/translatetext?' + params.join('&');

      GM_xmlhttpRequest({
        method: 'POST',
        url: url,
        headers: {
          'Content-Type': 'application/json',
          'Accept': '*/*'
        },
        data: JSON.stringify(texts),
        onload: res => {
          if (res.status === 200) {
            try {
              const result = JSON.parse(res.responseText);
              if (!Array.isArray(result)) return reject(new Error('返回格式异常'));
              resolve(result.map(item => item && item.translations && item.translations[0] ? item.translations[0].text : null));
            } catch (e) { reject(new Error('解析失败')); }
          } else {
            reject(new Error('翻译失败: ' + res.status));
          }
        },
        onerror: () => reject(new Error('网络请求失败'))
      });
    });
  }

  async function translateGoogleText(text, fromLang, toLang) {
    const targetLang = normalizeTargetLang(toLang, 'google');
    const sourceLang = normalizeSourceLang(fromLang, 'google') || 'auto';
    const url = 'https://translate.googleapis.com/translate_a/single?client=gtx&sl=' +
      encodeURIComponent(sourceLang) + '&tl=' + encodeURIComponent(targetLang) +
      '&dt=t&q=' + encodeURIComponent(text);
    const response = await requestText({ url: url });
    const result = JSON.parse(response);
    if (!Array.isArray(result) || !Array.isArray(result[0])) return null;
    return result[0].map(function(part) {
      return Array.isArray(part) ? part[0] : '';
    }).join('');
  }

  async function translateMyMemoryText(text, fromLang, toLang) {
    const targetLang = normalizeTargetLang(toLang, 'mymemory');
    const sourceLang = normalizeSourceLang(fromLang, 'mymemory') || 'auto';
    const url = 'https://api.mymemory.translated.net/get?q=' + encodeURIComponent(text) +
      '&langpair=' + encodeURIComponent(sourceLang + '|' + targetLang);
    const response = await requestText({ url: url });
    const result = JSON.parse(response);
    return result && result.responseData ? result.responseData.translatedText : null;
  }

  function getValueByPath(source, path) {
    if (!path) return source;
    return path.split('.').reduce(function(value, key) {
      if (value === null || value === undefined) return undefined;
      if (/^\d+$/.test(key)) return value[Number(key)];
      return value[key];
    }, source);
  }

  async function translateCustomText(text, fromLang, toLang) {
    const template = translatorSettings.customUrl.trim();
    if (!template) return null;

    const replacements = {
      text: encodeURIComponent(text),
      from: encodeURIComponent(normalizeSourceLang(fromLang, 'custom') || 'auto'),
      to: encodeURIComponent(normalizeTargetLang(toLang, 'custom'))
    };
    let url = template.replace(/\{(text|from|to)\}/g, function(match, key) {
      return replacements[key];
    });

    if (url.indexOf('{text}') === -1 && template.indexOf('{text}') === -1) {
      url += (url.indexOf('?') === -1 ? '?' : '&') + 'text=' + replacements.text;
    }

    const response = await requestText({ url: url });
    try {
      const parsed = JSON.parse(response);
      const value = getValueByPath(parsed, translatorSettings.customResultPath);
      return value === undefined || value === null ? null : String(value);
    } catch (e) {
      return response;
    }
  }

  async function translateOneByOne(texts, translateText, shouldContinue) {
    const results = [];
    for (const text of texts) {
      if (typeof shouldContinue === 'function' && !shouldContinue()) break;
      try {
        results.push(await translateText(text));
      } catch (e) {
        console.error('单条翻译失败:', e);
        results.push(null);
      }
      await new Promise(function(resolve) { setTimeout(resolve, 30); });
    }
    return results;
  }

  function translateBatch(texts, fromLang, toLang, shouldContinue) {
    const engine = translatorSettings.engine;
    if (engine === 'microsoft') return translateMicrosoftBatch(texts, normalizeSourceLang(fromLang, engine), normalizeTargetLang(toLang, engine));
    if (engine === 'google') return translateOneByOne(texts, function(text) {
      return translateGoogleText(text, fromLang, toLang);
    }, shouldContinue);
    if (engine === 'mymemory') return translateOneByOne(texts, function(text) {
      return translateMyMemoryText(text, fromLang, toLang);
    }, shouldContinue);
    if (engine === 'custom') return translateOneByOne(texts, function(text) {
      return translateCustomText(text, fromLang, toLang);
    }, shouldContinue);
    return Promise.reject(new Error('未知翻译服务'));
  }

  function getTranslationCacheKey(text) {
    const customKey = translatorSettings.engine === 'custom'
      ? translatorSettings.customUrl + '|' + translatorSettings.customResultPath
      : '';
    return translatorSettings.engine + '|' + customKey + '|' +
      languageSettings.from + '>' + languageSettings.to + '|' + text;
  }

  // ── 文本节点收集 ──────────────────────────────────────

  const EXCLUDED_SELECTORS = [
    'SCRIPT', 'STYLE', 'NOSCRIPT',
    '.microsoft-translated', '[data-no-translate]', '.no-translate', '.script-element',
    '[data-protect-translation="true"]',
    '#auto-translate-control', '#auto-translate-selection', '#auto-translate-subtitle',
    '.auto-translate-control', '.auto-translate-selection', '.auto-translate-subtitle',
    '[data-auto-translate-ui="true"]', '[data-auto-translate-selection-ui="true"]',
    '.ytp-caption-window-container', '.bpx-player-subtitle-panel', '.bilibili-player-video-subtitle',
    '.bili-subtitle-x-subtitle-panel', '.bpx-player-subtitle-wrap'
  ];

  function makeNodeFilter() {
    return {
      acceptNode: function(node) {
        const parent = node.parentElement;
        if (!parent) return NodeFilter.FILTER_REJECT;

        for (const sel of EXCLUDED_SELECTORS) {
          try { if (parent.matches(sel) || parent.closest(sel)) return NodeFilter.FILTER_REJECT; } catch (e) {}
        }

        if (parent.getAttribute('translate') === 'no') return NodeFilter.FILTER_REJECT;
        if (parent.classList.contains('no-translate') || parent.classList.contains('script-element')) return NodeFilter.FILTER_REJECT;
        if (parent.classList.contains('microsoft-translated')) return NodeFilter.FILTER_REJECT;

        const text = node.textContent.trim();
        if (!text || text.length < 2) return NodeFilter.FILTER_REJECT;
        if (/^\d+$/.test(text)) return NodeFilter.FILTER_REJECT;
        if (!/[a-zA-Z\u0400-\u04FF\u4e00-\u9fff]/.test(text)) return NodeFilter.FILTER_REJECT;

        return NodeFilter.FILTER_ACCEPT;
      }
    };
  }

  function collectTextNodes() {
    const textNodes = [];
    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, makeNodeFilter());
    let node;
    while ((node = walker.nextNode())) {
      const text = node.textContent.trim();
      if (text) textNodes.push({ node: node, text: text, parent: node.parentElement, originalText: node.textContent });
    }
    return textNodes;
  }

  // ── 批次划分 ──────────────────────────────────────────

  function createTextBatches(textNodes) {
    const batches = [];
    let current = { nodes: [], totalLength: 0 };

    for (const info of textNodes) {
      const len = info.text.length;
      if (len > MAX_TEXT_LENGTH) {
        if (current.nodes.length > 0) { batches.push(current); current = { nodes: [], totalLength: 0 }; }
        const truncated = Object.assign({}, info, { text: info.text.substring(0, MAX_TEXT_LENGTH) });
        batches.push({ nodes: [truncated], totalLength: MAX_TEXT_LENGTH });
        continue;
      }
      if (current.totalLength + len > MAX_TEXT_LENGTH || current.nodes.length >= BATCH_SIZE) {
        batches.push(current);
        current = { nodes: [], totalLength: 0 };
      }
      current.nodes.push(info);
      current.totalLength += len;
    }
    if (current.nodes.length > 0) batches.push(current);
    return batches;
  }

  // ── 应用翻译 ──────────────────────────────────────────

  function generateElementId() {
    return 'translated_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);
  }

  function applyTranslation(nodeInfo, translation) {
    try {
      const originalNode = nodeInfo.node;
      const parent = nodeInfo.parent;
      const originalText = nodeInfo.originalText;

      const elementId = generateElementId();
      originalTextMap.set(elementId, originalText);

      const span = document.createElement('span');
      span.className = 'microsoft-translated no-translate script-element';
      span.dataset.originalId = elementId;
      span.setAttribute('translate', 'no');
      span.setAttribute('data-no-translate', 'true');
      span.setAttribute('data-protect-translation', 'true');
      span.textContent = translation;

      parent.replaceChild(span, originalNode);
    } catch (e) {
      console.error('应用翻译失败:', e);
    }
  }

  // ── 翻译执行 ──────────────────────────────────────────

  async function runTranslation(textNodes) {
    if (!translationEnabled || !translatorReady || textNodes.length === 0) return;
    const fromLang = languageSettings.from === 'auto' ? null : languageSettings.from;
    const targetLang = getTranslationTargetLang();
    const batches = createTextBatches(textNodes);

    for (let i = 0; i < batches.length; i++) {
      if (!translationEnabled || !pageTranslating) break;
      const batch = batches[i];
      const toTranslate = [];
      const indices = [];

      batch.nodes.forEach(function(info, idx) {
        const key = getTranslationCacheKey(info.text);
        if (translationCache.has(key)) {
          applyTranslation(info, translationCache.get(key));
        } else {
          toTranslate.push(info.text);
          indices.push(idx);
        }
      });

      if (toTranslate.length > 0) {
        try {
          const translations = await translateBatch(toTranslate, fromLang, targetLang, function() {
            return translationEnabled && pageTranslating;
          });
          if (!translationEnabled || !pageTranslating) break;
          translations.forEach(function(tr, ti) {
            const info = batch.nodes[indices[ti]];
            if (tr && tr !== info.text) {
              applyTranslation(info, tr);
              translationCache.set(getTranslationCacheKey(info.text), tr);
            }
          });
        } catch (e) {
          console.error('批次 ' + (i + 1) + ' 翻译失败:', e);
          if (translationRetryCount < MAX_RETRY_COUNT) {
            translationRetryCount++;
            await new Promise(function(r) { setTimeout(r, 200); });
            i--;
            continue;
          }
        }
      }

      if (i < batches.length - 1) {
        await new Promise(function(r) { setTimeout(r, 20); });
      }
    }
  }

  async function translateWholePage() {
    if (!translationEnabled || pageTranslating || !translatorReady) return;
    pageTranslating = true;
    translationRetryCount = 0;
    try {
      const textNodes = collectTextNodes();
      await runTranslation(textNodes);
      lastTranslationTime = Date.now();
      setTimeout(function() { checkForUntranslatedText(); }, 500);
    } catch (e) {
      console.error('页面翻译失败:', e);
    } finally {
      pageTranslating = false;
    }
  }

  function checkForUntranslatedText() {
    if (!translationEnabled || !translatorReady || pageTranslating) return;
    const nodes = collectTextNodes();
    if (nodes.length === 0) return;
    if (nodes.length > 10) {
      translateWholePage();
    } else {
      (async function() {
        if (pageTranslating) return;
        pageTranslating = true;
        try { await runTranslation(nodes); } finally { pageTranslating = false; }
      })();
    }
  }

  // ── DOM 监听 ──────────────────────────────────────────

  function isAutoTranslateUiNode(node) {
    if (!node) return false;
    const element = node.nodeType === 1 ? node : node.parentElement;
    return !!(element && element.closest && element.closest('[data-auto-translate-ui="true"]'));
  }

  function mutationAffectsPage(mutation) {
    if (isAutoTranslateUiNode(mutation.target)) return false;

    if (mutation.type === 'characterData') {
      return mutation.target.nodeType === Node.TEXT_NODE;
    }

    if (mutation.type === 'childList') {
      const nodes = Array.prototype.slice.call(mutation.addedNodes).concat(Array.prototype.slice.call(mutation.removedNodes));
      if (nodes.length === 0) return false;
      return nodes.some(function(node) {
        return !isAutoTranslateUiNode(node);
      });
    }

    return false;
  }

  function startDOMObservation() {
    if (isObservingDOM || !translationEnabled || !translatorReady) return;
    stopDOMObservation();
    domObserver = new MutationObserver(function(mutations) {
      if (pageTranslating) return;
      if (domChangeTimer) clearTimeout(domChangeTimer);
      domChangeTimer = setTimeout(function() {
        const hasChanges = mutations.some(mutationAffectsPage);
        if (hasChanges && translationEnabled) setTimeout(function() { translateWholePage(); }, 300);
      }, DOM_CHANGE_DELAY);
    });
    try {
      domObserver.observe(document.body, { childList: true, subtree: true, characterData: true });
      isObservingDOM = true;
    } catch (e) { console.error('DOM观察器启动失败:', e); }
  }

  function stopDOMObservation() {
    if (domObserver) { domObserver.disconnect(); domObserver = null; }
    if (domChangeTimer) { clearTimeout(domChangeTimer); domChangeTimer = null; }
    isObservingDOM = false;
  }

  // ── 持续翻译 & 滚动检测 ───────────────────────────────

  function startContinuousTranslation() {
    if (continuousTranslationTimer) clearInterval(continuousTranslationTimer);
    continuousTranslationTimer = setInterval(function() {
      if (translationEnabled && !pageTranslating && translatorReady && Date.now() - lastTranslationTime > 20000) {
        checkForUntranslatedText();
      }
    }, 8000);
    lastScrollY = window.scrollY;
    window.addEventListener('scroll', handleScroll);
  }

  function stopContinuousTranslation() {
    if (continuousTranslationTimer) {
      clearInterval(continuousTranslationTimer);
      continuousTranslationTimer = null;
    }
    if (scrollCheckTimer) {
      clearTimeout(scrollCheckTimer);
      scrollCheckTimer = null;
    }
    window.removeEventListener('scroll', handleScroll);
  }

  function handleScroll() {
    if (Math.abs(window.scrollY - lastScrollY) > 300) {
      lastScrollY = window.scrollY;
      if (scrollCheckTimer) clearTimeout(scrollCheckTimer);
      scrollCheckTimer = setTimeout(function() {
        if (translationEnabled && !pageTranslating && translatorReady) checkForUntranslatedText();
      }, SCROLL_CHECK_DELAY);
    }
  }

  function isSelectionUiTarget(target) {
    const element = target && target.nodeType === 1 ? target : (target && target.parentElement);
    return !!(element && element.closest && element.closest('[data-auto-translate-selection-ui="true"]'));
  }

  function getSelectionAnchorElement(selection) {
    const node = selection.anchorNode;
    if (!node) return null;
    return node.nodeType === 1 ? node : node.parentElement;
  }

  function isEditableSelection(selection) {
    const element = getSelectionAnchorElement(selection);
    if (!element) return false;
    if (element.isContentEditable) return true;
    if (!element.closest) return false;
    return !!element.closest('input, textarea, select');
  }

  function getCurrentSelectionInfo() {
    if (!selectionTranslateEnabled) return null;

    const selection = window.getSelection();
    if (!selection || selection.rangeCount === 0 || selection.isCollapsed) return null;

    const anchorElement = getSelectionAnchorElement(selection);
    if (anchorElement && anchorElement.closest && anchorElement.closest('[data-auto-translate-ui="true"]')) return null;
    if (isEditableSelection(selection)) return null;

    const fullText = selection.toString().trim();
    if (!fullText) return null;

    let rect = null;
    try {
      const range = selection.getRangeAt(0);
      const rects = range.getClientRects();
      rect = rects.length > 0 ? rects[rects.length - 1] : range.getBoundingClientRect();
    } catch (e) {
      rect = null;
    }
    if (!rect || (!rect.width && !rect.height)) return null;

    return {
      text: fullText.length > SELECTION_MAX_LENGTH ? fullText.slice(0, SELECTION_MAX_LENGTH) : fullText,
      truncated: fullText.length > SELECTION_MAX_LENGTH,
      rect: rect
    };
  }

  function updateSelectionUiVisibility() {
    if (!selectionUiRoot) return;
    const toolbar = selectionUiRoot.querySelector('[data-role="selection-toolbar"]');

    const hasVisible = (toolbar && !toolbar.hidden) || (result && !result.hidden);
    selectionUiRoot.hidden = !hasVisible;
  }

  function positionSelectionFloating(element, rect, preferAbove) {
    if (!element) return;

    const margin = 8;
    const gap = 8;
    if (selectionUiRoot) selectionUiRoot.hidden = false;
    element.style.visibility = 'hidden';
    element.hidden = false;

    const width = element.offsetWidth || 90;
    const height = element.offsetHeight || 36;
    let left = rect.left + rect.width / 2 - width / 2;
    let top = preferAbove ? rect.top - height - gap : rect.bottom + gap;

    if (preferAbove && top < margin) top = rect.bottom + gap;
    if (!preferAbove && top + height > window.innerHeight - margin) top = rect.top - height - gap;

    left = Math.min(Math.max(margin, left), Math.max(margin, window.innerWidth - width - margin));
    top = Math.min(Math.max(margin, top), Math.max(margin, window.innerHeight - height - margin));

    element.style.left = Math.round(left) + 'px';
    element.style.top = Math.round(top) + 'px';
    element.style.visibility = '';
  }

  function showSelectionToolbarForCurrentSelection() {
    if (!selectionTranslateEnabled) return;

    const info = getCurrentSelectionInfo();
    if (!info) {
      hideSelectionToolbar();
      return;
    }

    if (selectionResultOpen) {
      if (info.text === selectionText) return;
      hideSelectionResult();
    }

    selectionText = info.text;
    selectionTextTruncated = info.truncated;
    selectionAnchorRect = info.rect;
    ensureSelectionUi();
    if (!selectionUiRoot) return;

    const toolbar = selectionUiRoot.querySelector('[data-role="selection-toolbar"]');
    if (!toolbar) return;

    positionSelectionFloating(toolbar, info.rect, true);
    updateSelectionUiVisibility();
  }

    if (selectionUiRoot) {
      const toolbar = selectionUiRoot.querySelector('[data-role="selection-toolbar"]');
      if (toolbar) toolbar.hidden = true;
    }
    updateSelectionUiVisibility();
  }

    selectionRequestId++;

    stopSelectionSpeech();
    if (selectionUiRoot) {
      const result = selectionUiRoot.querySelector('[data-role="selection-result"]');
      if (result) result.hidden = true;

  }

  function toSpeechLang(code) {
    const normalized = normalizeDetectedLanguageCode(code);
    return SPEECH_LANG_MAP[normalized] || normalized || navigator.language || 'en-US';
  }

  function stopSelectionSpeech() {
    if (selectionSpeakingText && isSpeechSupported()) {
      try { window.speechSynthesis.cancel(); } catch (e) {}
    }
    selectionSpeakingText = '';
  }

  function speakSelectionText(text, lang) {
    if (!text || !isSpeechSupported()) return;

    const synth = window.speechSynthesis;
    if (selectionSpeakingText === text && (synth.speaking || synth.pending)) {
      stopSelectionSpeech();
      return;
    }

    stopSelectionSpeech();
    const utterance = new SpeechSynthesisUtterance(text);
    utterance.lang = toSpeechLang(lang);
    utterance.onend = function() {
      if (selectionSpeakingText === text) selectionSpeakingText = '';
    };
    utterance.onerror = function() {
      if (selectionSpeakingText === text) selectionSpeakingText = '';
    };
    selectionSpeakingText = text;
    try {
      synth.speak(utterance);
    } catch (e) {
      selectionSpeakingText = '';
    }
  }

  function speakSelectedText() {
    if (!selectionText) return;
    const detected = detectSelectionLanguageByScript(selectionText, selectionDirection.langA, selectionDirection.langB);
    const pageLang = document.documentElement
      ? normalizeDetectedLanguageCode(document.documentElement.getAttribute('lang'))
      : null;
    speakSelectionText(selectionText, detected || pageLang || selectionDirection.langA);
  }

    if (!result || !title || !body) return;

    if (state !== 'done') stopSelectionSpeech();
    selectionResultOpen = true;
    result.dataset.state = state;
    if (state === 'done') {
      const sourceLabel = getLanguageLabel(selectionResultSourceLang);
      const targetLabel = getLanguageLabel(selectionResultTargetLang);
      title.textContent = sourceLabel && targetLabel
        ? '翻译结果 · ' + sourceLabel + ' → ' + targetLabel
        : (targetLabel ? '翻译结果 · 翻译为' + targetLabel : '翻译结果');
    } else {
      title.textContent = state === 'loading' ? '正在翻译' : '翻译失败';
    }
    body.textContent = content;
    if (note) note.hidden = !(state === 'done' && selectionTextTruncated);
    if (speakButton) speakButton.hidden = !(state === 'done' && isSpeechSupported());

      left: window.innerWidth / 2,

      width: 0,
      height: 0,
      bottom: window.innerHeight / 3

    positionSelectionFloating(result, rect, false);
    updateSelectionUiVisibility();
  }

  function getLanguageScript(lang) {
    return LANGUAGE_SCRIPT_MAP[lang] || 'latin';
  }

  function detectTextScript(text) {
    const sample = String(text || '').slice(0, 500);
    if (/[぀-ヿ]/.test(sample)) return 'kana';
    if (/[가-힯]/.test(sample)) return 'hangul';

    let bestScript = null;
    let bestCount = 0;
    for (let i = 0; i < SCRIPT_PATTERNS.length; i++) {
      let count = (sample.match(SCRIPT_PATTERNS[i][1]) || []).length;
      // 一个汉字的信息量远大于一个拉丁字母，加权避免中英混排被误判成英文。
      if (SCRIPT_PATTERNS[i][0] === 'han') count *= 3;
      if (count > bestCount) {
        bestCount = count;
        bestScript = SCRIPT_PATTERNS[i][0];
      }
    }
    return bestCount > 0 ? bestScript : null;
  }

  function detectSelectionLanguageByScript(text, langA, langB) {
    const script = detectTextScript(text);
    if (!script) return null;

    const scriptA = getLanguageScript(langA);
    const scriptB = getLanguageScript(langB);
    const matchA = scriptA === script;
    const matchB = scriptB === script;
    if (matchA && !matchB) return langA;
    if (matchB && !matchA) return langB;
    return null;
  }

  function normalizeDetectedLanguageCode(code) {
    if (!code) return null;
    const value = String(code).toLowerCase();
    if (value === 'zh' || value === 'zh-cn' || value === 'zh-sg' || value === 'zh-hans') return 'zh-Hans';
    if (value === 'zh-tw' || value === 'zh-hk' || value === 'zh-hant') return 'zh-Hant';
    return value.split('-')[0];
  }

  function languageCodesMatch(a, b) {
    if (!a || !b) return false;
    const valueA = String(a).toLowerCase();
    const valueB = String(b).toLowerCase();
    if (valueA === valueB) return true;
    const baseA = valueA.split('-')[0];
    const baseB = valueB.split('-')[0];
    if (baseA !== baseB) return false;
    if (baseA === 'zh' && valueA.indexOf('-') !== -1 && valueB.indexOf('-') !== -1) return false;
    return true;
  }

  function getSelectionDirectionTarget(detectedLang, langA, langB, fallbackTarget) {
    if (detectedLang) {
      if (languageCodesMatch(detectedLang, langA)) return langB;
      if (languageCodesMatch(detectedLang, langB)) return langA;
    }
    return fallbackTarget;
  }

  function translateMicrosoftSelection(text, toLang) {
    return new Promise(function(resolve, reject) {
      const url = 'https://edge.microsoft.com/translate/translatetext?to=' + encodeURIComponent(toLang);
      GM_xmlhttpRequest({
        method: 'POST',
        url: url,
        headers: {
          'Content-Type': 'application/json',
          'Accept': '*/*'
        },
        data: JSON.stringify([text]),
        onload: function(res) {
          if (res.status === 200) {
            try {
              const result = JSON.parse(res.responseText);
              const item = Array.isArray(result) ? result[0] : null;
              const translation = item && item.translations && item.translations[0] ? item.translations[0].text : null;
              const detectedLang = item && item.detectedLanguage ? item.detectedLanguage.language : null;
              resolve({ translation: translation, detectedLang: detectedLang });
            } catch (e) {
              reject(new Error('解析失败'));
            }
          } else {
            reject(new Error('翻译失败: ' + res.status));
          }
        },
        onerror: function() { reject(new Error('网络请求失败')); }
      });
    });
  }

  async function translateGoogleSelection(text, toLang) {
    const targetLang = normalizeTargetLang(toLang, 'google');
    const url = 'https://translate.googleapis.com/translate_a/single?client=gtx&sl=auto&tl=' +
      encodeURIComponent(targetLang) + '&dt=t&q=' + encodeURIComponent(text);
    const response = await requestText({ url: url });
    const result = JSON.parse(response);
    const translation = Array.isArray(result) && Array.isArray(result[0])
      ? result[0].map(function(part) { return Array.isArray(part) ? part[0] : ''; }).join('')
      : null;
    const detectedLang = Array.isArray(result) && typeof result[2] === 'string' ? result[2] : null;
    return { translation: translation, detectedLang: detectedLang };
  }

  async function translateSelectionText(text) {
    await setupActiveTranslator();
    if (!translatorReady) throw new Error('翻译服务不可用');

    // 划词翻译使用自己的语言 B 作为目标，不再跟随悬浮翻译的目标语言。
    const fallbackTarget = selectionDirection.langB;
    if (!selectionDirection.smartDetect) {
      const results = await translateBatch([text], 'auto', fallbackTarget);
      return makeSelectionResult(results && results[0], '', fallbackTarget);
    }

    const langA = selectionDirection.langA;
    const langB = selectionDirection.langB;
    const localLang = detectSelectionLanguageByScript(text, langA, langB);

    if (localLang) {
      const target = localLang === langA ? langB : langA;
      const results = await translateBatch([text], 'auto', target);
      return makeSelectionResult(results && results[0], localLang, target);
    }

    const engine = translatorSettings.engine;
    if (engine === 'microsoft' || engine === 'google') {
      const probeTarget = langB;
      const probe = engine === 'microsoft'
        ? await translateMicrosoftSelection(text, probeTarget)
        : await translateGoogleSelection(text, probeTarget);
      const detected = normalizeDetectedLanguageCode(probe.detectedLang);
      const target = getSelectionDirectionTarget(detected, langA, langB, fallbackTarget);

      if (target === probeTarget && probe.translation) {
        return makeSelectionResult(probe.translation, detected || '', target);
      }
      const results = await translateBatch([text], 'auto', target);
      return makeSelectionResult(results && results[0], detected || '', target);
    }

    const results = await translateBatch([text], 'auto', langB);
    return makeSelectionResult(results && results[0], '', langB);
  }

  async function translateCurrentSelection() {
    const text = selectionText;
    if (!text) return;

    selectionIgnoreHideUntil = Date.now() + 2000;
    hideSelectionToolbar();
    showSelectionResult('loading', '正在翻译…');
    const requestId = ++selectionRequestId;

    try {
      const result = await translateSelectionText(text);
      if (requestId !== selectionRequestId) return;
      if (!result || !result.translation) throw new Error('没有返回翻译结果');
      selectionResultSourceLang = result.sourceLang;
      selectionResultTargetLang = result.targetLang;
      showSelectionResult('done', result.translation);
    } catch (error) {
      if (requestId !== selectionRequestId) return;
      const message = error && error.message ? error.message : String(error);
      showSelectionResult('error', '翻译失败：' + message + '\n\n可在脚本猫的脚本配置里更换翻译服务。');
    }
  }

  function scheduleSelectionToolbar(delay) {
    if (!selectionTranslateEnabled) return;

    selectionShowTimer = setTimeout(function() {
      selectionShowTimer = null;

  function handleSelectionMouseUp(event) {
    selectionPointerDown = false;
    if (event.button !== undefined && event.button !== 0) return;
    if (isSelectionUiTarget(event.target)) return;
    scheduleSelectionToolbar(20);
  }

  function handleSelectionTouchEnd(event) {
    selectionPointerDown = false;
    if (isSelectionUiTarget(event.target)) return;
    scheduleSelectionToolbar(150);
  }

  function handleSelectionKeyUp(event) {
    if (event.key === 'Escape') return;
    if (isSelectionUiTarget(event.target)) return;
    const selection = window.getSelection();
    if (!selection || selection.isCollapsed) return;
    scheduleSelectionToolbar(120);
  }

  function handleSelectionChange() {
    if (!selectionTranslateEnabled || selectionPointerDown || selectionResultOpen) return;

    const selection = window.getSelection();
    if (!selection || selection.isCollapsed || !selection.toString().trim()) {
      if (Date.now() > selectionIgnoreHideUntil) hideSelectionToolbar();
      return;
    }
    if (selectionShowTimer) return;
    scheduleSelectionToolbar(350);
  }

  function handleSelectionPointerDown(event) {
    selectionPointerDown = true;
    if (isSelectionUiTarget(event.target)) return;

    if (selectionShowTimer) {
      clearTimeout(selectionShowTimer);
      selectionShowTimer = null;
    }
    hideSelectionUi();
  }

  function handleSelectionScroll() {
    hideSelectionToolbar();
  }

  function handleSelectionResize() {
    hideSelectionToolbar();
  }

  function handleSelectionKeyDown(event) {
    if (event.key === 'Escape') hideSelectionUi();
  }

  function attachSelectionListeners() {
    if (selectionListenersAttached) return;
    selectionListenersAttached = true;
    document.addEventListener('mouseup', handleSelectionMouseUp, true);
    document.addEventListener('touchend', handleSelectionTouchEnd, true);
    document.addEventListener('keyup', handleSelectionKeyUp, true);
    document.addEventListener('keydown', handleSelectionKeyDown, true);
    document.addEventListener('mousedown', handleSelectionPointerDown, true);
    document.addEventListener('touchstart', handleSelectionPointerDown, true);
    document.addEventListener('selectionchange', handleSelectionChange);
    document.addEventListener('scroll', handleSelectionScroll, true);
    window.addEventListener('resize', handleSelectionResize);
  }

  function isSubtitlePage(adapter) {
    if (!adapter) return false;
    if (adapter.id === 'youtube') {
      return /^\/(?:watch(?:\/|$)|shorts\/|embed\/|live\/|v\/|playlist(?:\/|$))/.test(location.pathname);

    if (adapter.id === 'bilibili') {
      return /^\/(video|bangumi|live|list|festival|medialist|cheese)\//.test(location.pathname) ||
        location.pathname === '/player.html';
    }

  }

    return String(text || '').replace(/\s+/g, ' ').trim();

    if (typeof element.innerText === 'string') return element.innerText;
    return element.textContent || '';

    if (adapter && typeof adapter.getText === 'function') return adapter.getText(element);

  function isSubtitleContainerVisible(element) {
    const style = getComputedStyle(element);
    return style.display !== 'none' && style.visibility !== 'hidden' && element.getClientRects().length > 0;
  }

  function findSubtitleContainer(adapter) {
    if (!adapter) return null;
    for (let i = 0; i < adapter.selectors.length; i++) {
      let best = null;
      let bestLength = 0;
      const elements = document.querySelectorAll(adapter.selectors[i]);
      for (let j = 0; j < elements.length; j++) {
        const element = elements[j];
        if (!isSubtitleContainerVisible(element)) continue;
        const text = normalizeSubtitleText(getSubtitleAdapterText(adapter, element));
        if (text.length > bestLength) {
          best = element;
          bestLength = text.length;
        }
      }
      if (best) return best;
    }
    return null;
  }

  function getSubtitleCacheKey(text) {
    const customKey = translatorSettings.engine === 'custom'
      ? translatorSettings.customUrl + '|' + translatorSettings.customResultPath
      : '';

  function cacheSubtitleTranslation(key, translation) {
    subtitleCache.set(key, translation);
    if (subtitleCache.size > 800) {
      const firstKey = subtitleCache.keys().next().value;
      subtitleCache.delete(firstKey);
    }
  }

  function scheduleSubtitleTranslation(text, requestId, immediate) {
    if (subtitleRequestInFlight) {
      subtitlePendingRequest = { text: text, requestId: requestId };
      return;
    }

    // 新的一句立即翻译；同一句继续变长时保留已经排好的定时器，
    // 只更新待翻译文本，避免字幕每几十毫秒更新一次时定时器被不断重置。
    if (immediate && subtitleTranslateTimer) {
      clearTimeout(subtitleTranslateTimer);
      subtitleTranslateTimer = null;
    }

    subtitlePendingRequest = { text: text, requestId: requestId };
    if (subtitleTranslateTimer) return;

    subtitleTranslateTimer = setTimeout(function() {
      subtitleTranslateTimer = null;
      const pending = subtitlePendingRequest;
      subtitlePendingRequest = null;
      if (!pending || pending.requestId !== subtitleRequestId) return;
      translateSubtitleText(pending.text, pending.requestId);
    }, immediate ? 0 : SUBTITLE_TRANSLATE_DELAY);
  }

  async function translateSubtitleText(text, requestId) {
    if (requestId !== subtitleRequestId) return;

    const cached = subtitleCache.get(getSubtitleCacheKey(text));
    if (cached) {
      subtitleLastTranslation = cached;
      renderSubtitleOverlay(text, cached, false);
      return;
    }

    if (subtitleRequestInFlight) {
      subtitlePendingRequest = { text: text, requestId: requestId };
      return;
    }

    subtitleRequestInFlight = true;
    try {
      const results = await translateBatch([text], 'auto', subtitleTargetLang);
      const translation = results && results[0];
      if (translation) {
        cacheSubtitleTranslation(getSubtitleCacheKey(text), translation);
        if (requestId === subtitleRequestId) {
          subtitleLastTranslation = translation;
          renderSubtitleOverlay(text, translation, false);
        }
      } else if (requestId === subtitleRequestId) {
        renderSubtitleOverlay(text, '', false);
      }
    } catch (error) {
      if (requestId === subtitleRequestId) renderSubtitleOverlay(text, '', false);
      console.error('字幕翻译失败:', error);
    } finally {
      subtitleRequestInFlight = false;
      const pending = subtitlePendingRequest;
      subtitlePendingRequest = null;
      if (pending && pending.requestId === subtitleRequestId && pending.requestId !== requestId) {
        translateSubtitleText(pending.text, pending.requestId);
      }
    }
  }

  function updateSubtitleTranslation() {
    if (!subtitleEnabled) return;

    subtitleAdapter = getSubtitleAdapter();
    if (!subtitleAdapter || !isSubtitlePage(subtitleAdapter)) {
      clearSubtitleState();
      return;
    }

    const container = findSubtitleContainer(subtitleAdapter);
    if (!container) {
      clearSubtitleState();
      return;
    }

    const text = normalizeSubtitleText(getSubtitleAdapterText(subtitleAdapter, container));
    if (!text) {
      clearSubtitleState();
      return;
    }

    const playerRoot = getSubtitlePlayerRoot(container);
    if (playerRoot && playerRoot !== subtitlePlayerRoot) {
      subtitlePlayerRoot = playerRoot;
      if (subtitleOverlay) appendSubtitleOverlayToMount();
    }

    if (subtitleHiddenContainer && subtitleHiddenContainer !== container) {
      subtitleHiddenContainer.classList.remove('auto-translate-native-caption');
    }
    subtitleHiddenContainer = container;
    container.classList.add('auto-translate-native-caption');

    if (text === subtitleCurrentText) {
      ensureSubtitleOverlay();
      if (!subtitleOverlay) return;
      const originalElement = subtitleOverlay.querySelector('[data-role="subtitle-original"]');
      if (originalElement && originalElement.textContent !== text) {
        const cached = subtitleCache.get(getSubtitleCacheKey(text));
        if (cached) subtitleLastTranslation = cached;
        renderSubtitleOverlay(text, cached || '', false);
      } else {
        subtitleOverlay.hidden = false;
      }
      return;
    }

    const previousText = subtitleCurrentText;
    const isContinuation = !!previousText && text.indexOf(previousText) === 0;
    if (!isContinuation) subtitleLastTranslation = '';

    subtitleCurrentText = text;
    const requestId = ++subtitleRequestId;
    renderSubtitleOverlay(text, isContinuation ? subtitleLastTranslation : '', true);

    const cached = subtitleCache.get(getSubtitleCacheKey(text));
    if (cached) {
      if (subtitleTranslateTimer) {
        clearTimeout(subtitleTranslateTimer);
        subtitleTranslateTimer = null;
      }
      subtitlePendingRequest = null;
      subtitleLastTranslation = cached;
      renderSubtitleOverlay(text, cached, false);
      return;
    }

    scheduleSubtitleTranslation(text, requestId, !isContinuation);
  }

  function scheduleSubtitleUpdate() {
    if (subtitleUpdateTimer) return;
    subtitleUpdateTimer = setTimeout(function() {
      subtitleUpdateTimer = null;
      updateSubtitleTranslation();
    }, 60);
  }

  function handleSubtitleFullscreenChange() {
    appendSubtitleOverlayToMount();
  }

  function stopSubtitleObserver() {
    if (subtitleObserver) {
      subtitleObserver.disconnect();
      subtitleObserver = null;
    }
    if (subtitlePollTimer) {
      clearInterval(subtitlePollTimer);
      subtitlePollTimer = null;
    }
    if (subtitleUpdateTimer) {
      clearTimeout(subtitleUpdateTimer);
      subtitleUpdateTimer = null;
    }
    if (subtitleTranslateTimer) {
      clearTimeout(subtitleTranslateTimer);
      subtitleTranslateTimer = null;
    }
    document.removeEventListener('fullscreenchange', handleSubtitleFullscreenChange, true);
    document.removeEventListener('webkitfullscreenchange', handleSubtitleFullscreenChange, true);
  }

  function startSubtitleObserver() {
    stopSubtitleObserver();
    if (!document.body) return;

    if (typeof MutationObserver === 'function') {
      subtitleObserver = new MutationObserver(scheduleSubtitleUpdate);
      subtitleObserver.observe(document.body, {
        childList: true,
        subtree: true,
        characterData: true
      });
    }
    subtitlePollTimer = setInterval(updateSubtitleTranslation, 600);
    document.addEventListener('fullscreenchange', handleSubtitleFullscreenChange, true);
    document.addEventListener('webkitfullscreenchange', handleSubtitleFullscreenChange, true);
  }

  async function startSubtitleTranslation() {
    if (!subtitleEnabled) return;

    subtitleAdapter = getSubtitleAdapter();
    if (!subtitleAdapter) return;

    await setupActiveTranslator();
    if (!subtitleEnabled || !translatorReady) return;

    startSubtitleObserver();
    ensureSubtitleOverlay();
    updateSubtitleTranslation();
  }

  // ── 初始化 ────────────────────────────────────────────

  async function init() {
    readTranslatorSettings();
    readLanguageSettings();
    readSelectionDirection();
    readIconOpacity();
    readDefaultTranslate();
    readFloatingTranslateEnabled();
    readSelectionTranslate();
    readSubtitleSettings();

    if (subtitleEnabled) {
      startSubtitleTranslation().catch(function(error) {
        console.error('视频字幕启动失败:', error);
      });
    }
    if (selectionTranslateEnabled) attachSelectionListeners();

    if (!floatingTranslateEnabled) {
      translationEnabled = false;
      return;
    }

    ensureControlUi();
    updateControlUi();

    if (defaultTranslateEnabled) {
      await enableTranslation();
    } else {
      translationEnabled = false;
    }
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }

})();
